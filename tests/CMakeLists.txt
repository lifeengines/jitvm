cmake_minimum_required(VERSION 3.16)

if(JITVM_HOST_X86 OR JITVM_HOST_X64)
	add_executable(jitvm_test jitvm_test.cpp)
	if(MSVC)
		if(JITVM_HOST_X64)
			target_sources(jitvm_test PRIVATE run_x64.asm)
		else()
			target_sources(jitvm_test PRIVATE run_x86.asm)
		endif()
		target_compile_options(jitvm_test PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHa>)
	else()
		if(JITVM_HOST_X64)
			target_sources(jitvm_test PRIVATE run_x64.S)
		else()
			target_sources(jitvm_test PRIVATE run_x86.S)
		endif()
	endif()
	target_link_libraries(jitvm_test PRIVATE jitvm)
	add_test(NAME jitvm_behavior COMMAND jitvm_test)
endif()
